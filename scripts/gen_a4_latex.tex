% The body font is monospace, whose inter-word glue has almost no natural
% stretch, so fully-justified lines otherwise end up with big gaps or overfull
% boxes. \emergencystretch lets TeX open up extra stretch to smooth line
% lengths, and the tolerance/badness limits keep "overfull" lines from turning
% into errors. Pair this with the CJK hanging punctuation above.

% Paragraph first-line indent: two full-width CJK characters (= 4 half-width
% spaces) on the monospace grid. The responsive package forces \parindent to
% 1em/1.5em in its own \AtBeginDocument hook, so this must run after it.
% 
% Body text is set ragged-right: the monospace font has almost no inter-word
% stretch, so fully-justified lines either leave big gaps or force aggressive
% hyphenation (e.g. breaking "Lovelock" as "Love-lock"). A small raggedness
% keeps the right edge mostly tidy while letting lines break naturally.

  \setlength{\parindent}{2em}%
  \RaggedRight

% Long digit runs with thousands separators (e.g. "2,400,000") are a single
% unbreakable Latin token and can overflow the column. Give the ASCII comma its
% own interchar class and plant a near-forbidden break after "comma before

% short numbers are untouched.
\newXeTeXintercharclass \postdigitcomma
\XeTeXcharclass `\, \postdigitcomma
\XeTeXinterchartoks \postdigitcomma 0 = {\penalty9990\relax}

% Article metadata block. Looks like a listings frame but is typeset in text
% mode: URLs are real hyperlinks (\url) that break anywhere via xurl, and an
% \fbox never cracks the way a listings frame does on wrapped lines. The
% generator emits this environment inside a ```{=tex} raw block.
% Wrapped continuation lines get a 2em hanging indent (\everypar because
% \hangindent resets after each paragraph).
\newsavebox{\postmetadatabox}
\newenvironment{postmetadata}{%
  \begin{lrbox}{\postmetadatabox}%
  \begin{minipage}{\dimexpr\postnarrowwidth-2\fboxsep-2\fboxrule\relax}%
  \footnotesize\raggedright
  \everypar{\hangindent=2em\hangafter=1}%
  % Match the inner top inset of the code blocks (fbox + listings): listings
  % struts its first line, which puts it ~2.5pt lower than an unstrutted text
  % line. Without this the metadata text hugs the top border while code text
  % does not. Calibrated 2026-09 against Sarasa Mono SC footnotesize.
  \vspace{2.5pt}%
}{%
  \end{minipage}%
  \end{lrbox}%
  \postblock{\fbox{\usebox{\postmetadatabox}}}%
}

% Inline code stays close to the listings look. Long literal tokens (paths,
% URLs, package names) contain no TeX line-break opportunities, so an inline
% code span wider than the column overflows. \postbreakables re-typesets the
% span while planting a near-forbidden break (\penalty 9990) after the
% punctuation characters that read as natural token boundaries; TeX only ever
% takes such a break to avoid an otherwise-overfull line, so normal short
% spans are untouched.
\ExplSyntaxOn
\int_new:N \l_postbreakpenalty_int
\int_set:Nn \l_postbreakpenalty_int { 9990 }
\tl_new:N \l_postbreakables_tl
\tl_new:N \l_post_search_tl
\tl_new:N \l_post_repl_tl
% Break opportunities are planted only after characters that read as natural
% token boundaries for paths/URLs. % # ~ are deliberately omitted (comment /
% special chars). TeX only takes these near-forbidden penalties to avoid an
% otherwise-overfull line, so ordinary short tokens are never split.
%
% The token list is transformed in place with \tl_replace_all rather than
% re-emitted token by token. A token mapper (\tl_map_inline) breaks whenever
% the content contains a macro that takes an argument -- e.g. a nested
% \markdownRendererEmphasis inside strong emphasis: the macro is emitted
% unexpanded, then swallows the next mapped item, raising "Argument of
% \__tl_map_1:w has an extra }". Replacing punctuation in place keeps every
% macro adjacent to its argument.
%
% \c_underscore_str / \c_ampersand_str are passed through \tl_set:Nx so the
% search tokens come out catcode 12, matching the markdown body (see
% \postmarkdowninput); a plain "_" or "&" here would have catcode 8/4 and
% never match the body.
\cs_new_protected:Npn \postbreakreplaceone #1 {
  \tl_set:Nx \l_post_search_tl { #1 }
  \tl_set_eq:NN \l_post_repl_tl \l_post_search_tl
  \tl_put_right:Nn \l_post_repl_tl { \penalty \l_postbreakpenalty_int }
  \exp_args:NNVV \tl_replace_all:Nnn \l_postbreakables_tl \l_post_search_tl \l_post_repl_tl
}
\cs_new_protected:Npn \postbreakreplace {
  \postbreakreplaceone { / }
  \postbreakreplaceone { . }
  \postbreakreplaceone { - }
  \postbreakreplaceone { \c_underscore_str }
  \postbreakreplaceone { : }
  \postbreakreplaceone { ? }
  \postbreakreplaceone { \c_ampersand_str }
  \postbreakreplaceone { = }
  \postbreakreplaceone { + }
}
\cs_new_protected:Npn \postbreakables #1 {
  \tl_set:Nn \l_postbreakables_tl { #1 }
  \postbreakreplace
  \tl_use:N \l_postbreakables_tl
}
\ExplSyntaxOff

% Notes are appended to \postnotes@acc in the SAME compile run and printed at
% the end by \postprintendnotes. This avoids the `endnotes` package, which ships
% markdown-rendered note content (links, %, #, ...) through an aux file whose
% line breaks / catcode changes break URLs. \long is required because note
% bodies may span multiple paragraphs (\par).
%
% Markers and entries are cross-linked: the in-text superscript [N] jumps to
% the endnote, the endnote's [N] jumps back to the in-text position. Named
% destinations are resolved by the PDF viewer, so a single run suffices.
\newcounter{postendnotecounter}
\makeatletter
\def\postnotes@acc{}
\long\def\postendnote#1{%
  \stepcounter{postendnotecounter}%
  % Bake the current number into the accumulator as plain digits; otherwise
  % \thepostendnotecounter would be expanded later, when all notes share the
  % final counter value.
  \edef\@postnotenum{\thepostendnotecounter}%
  \hypertarget{postnote-ref-\@postnotenum}{}%
  \textsuperscript{\hyperlink{postnote-def-\@postnotenum}{[\@postnotenum]}}%
  \expandafter\postendnote@add\expandafter{\@postnotenum}{#1}%
}
\long\def\postendnote@add#1#2{%
  \g@addto@macro\postnotes@acc{\par\noindent
    \hypertarget{postnote-def-#1}{}\hyperlink{postnote-ref-#1}{[#1]}\ #2}%
}
\newcommand{\postprintendnotes}{%
  \ifnum\value{postendnotecounter}>0
    % Separate endnotes from the body with the same gap a heading keeps from
    % the text above it, and give them the same narrow measure as
    % blockquotes (1em indent on each side).
    \par\addvspace{\baselineskip}%
    \noindent\hfill
    \begin{minipage}{\postquotewidth}%
    \postnotes@acc
    \end{minipage}%
    \hfill\null\par
  \fi
}
\makeatother

% Typography: H1-H3 are bold; H4-H6 (which markdown would otherwise turn
% into run-in \paragraph/\subparagraph heads that collide with the body
% text) are rendered as the same kind of left-aligned block headings at the
% base size, regular weight. Headings are \raggedright so a short title that
% wraps is not stretched across the line (which read as wide inter-character
% gaps); it breaks naturally instead. Every heading is distinguished by size
% and vertical spacing only. Sizes scale relative to the responsive base font so
% they remain balanced across layouts. The top gap is one line
% of the heading's OWN leading (switch to the scaled font first, then
% \baselineskip reflects the heading size), so a bigger heading gets a
% proportionally bigger break above it; the bottom gap stays a small
% fraction so the heading reads as belonging to the text below.
\newcommand{\postheadingone}[1]{%
  \par\begingroup
  \relscale{1.4}%
  \bfseries
  \raggedright
  \addvspace{\baselineskip}%
  \noindent#1\par
  \addvspace{0.3\baselineskip}%
  \endgroup
}
\newcommand{\postheadingtwo}[1]{%
  \par\begingroup
  \relscale{1.2}%
  \bfseries
  \raggedright
  \addvspace{\baselineskip}%
  \noindent#1\par
  \addvspace{0.25\baselineskip}%
  \endgroup
}
\newcommand{\postheadingthree}[1]{%

% Shared vertical gap around block-level elements (images, code blocks,
% blockquotes, the metadata frame, lists). The gap equals one *visible
% interline gap* (行距): the baseline pitch minus the vertical space the
% glyphs themselves occupy -- i.e. exactly the whitespace between two
% consecutive text lines. Measured at \begin{document} from a probe string
% with a cap-height letter and a descender, after the responsive package
% has fixed the font size (\baselineskip is only final there).
\newlength{\postblockgap}
\AtBeginDocument{%
  \setbox0=\hbox{Ag}%
  \setlength{\postblockgap}{\dimexpr\baselineskip-\ht0-\dp0\relax}%
}

% Uniform block placement: exactly \postblockgap of space above and below,
% measured from the element's outer box edge -- the frame edge for code
% blocks and the metadata frame -- not from the first/last baseline inside
% it. \raisebox{-\height} parks the box's top on the current line and
% \nointerlineskip keeps TeX's baseline glue from adding extra space.
\newcommand{\postblock}[1]{%
  \par\addvspace{\postblockgap}\nointerlineskip
  \noindent\hfill\raisebox{-\height}{#1}\hfill\null
  \par\addvspace{\postblockgap}\nointerlineskip
}

% Print a caption block only when #1 is non-empty.
% The caption text is left-aligned and spans the same width as the image
% above it (\linewidth inside the image minipage). The gap between image
% and caption is exactly one interline gap: \nointerlineskip kills TeX's
% baseline glue (which would add ~\baselineskip on top of \postblockgap),
% and the [t] minipage puts the caption's first-line top as the box top.
\newcommand{\postcaption}[1]{%
  \if\relax\detokenize{#1}\relax
  \else
    \nointerlineskip\vspace{\postblockgap}%
    \begin{minipage}[t]{\linewidth}%
      \raggedright
      \small
      #1%
    \end{minipage}%
    \par
  \fi
}

% Image renderer receives: #1=alt(caption), #2=escaped src, #3=raw src.
%
% Links are built from the *raw* URL (#3), with every TeX-special character
% escaped to the control sequence hyperref understands (\# \% \_ ...). Doing
% this at render time matters: footnote content travels through endnote aux
% files, where '#' (old hyperref: "Illegal parameter number in \Hy@tempa") and
% '%' would otherwise be re-interpreted with the wrong catcode and break.
% Image paths below keep #3 because \includegraphics needs the real filename.
\ExplSyntaxOn
\tl_new:N \l_posthref_url_tl
\cs_new_protected:Npn \posthref #1 #2 {
  % Work on a detokenized copy so '#'/special tokens keep their catcode
  % regardless of whether the URL came from the markdown input (catcode 12),
  % an endnote aux file (catcode 6), or plain text.
  \tl_clear:N \l_posthref_url_tl
  \tl_map_inline:Nn { \tl_to_str:n { #2 } } {
    \str_case:nnF { ##1 } {
      { \c_hash_str }       { \tl_put_right:Nn \l_posthref_url_tl { \# } }
      { \c_percent_str }    { \tl_put_right:Nn \l_posthref_url_tl { \% } }
      { \c_ampersand_str }  { \tl_put_right:Nn \l_posthref_url_tl { \& } }
      { \c_underscore_str } { \tl_put_right:Nn \l_posthref_url_tl { \_ } }
      { \c_left_brace_str } { \tl_put_right:Nn \l_posthref_url_tl { \{ } }
      { \c_right_brace_str }{ \tl_put_right:Nn \l_posthref_url_tl { \} } }
      { ~ } { \tl_put_right:Nn \l_posthref_url_tl { \textasciitilde {} } }
      { \c_circumflex_str } { \tl_put_right:Nn \l_posthref_url_tl { \^{} } }
    } {
      \tl_put_right:Nn \l_posthref_url_tl { ##1 }
    }
  }
  \href { \l_posthref_url_tl } { #1 }
}
\ExplSyntaxOff

\markdownSetup{
  renderers = {
    % markdown.sty is loaded before enumitem, so its theme never sees the
    % enumitem branch: tight lists would fall back to paralist's compactitem

    olBeginTight = {\begin{enumerate}},
    olEndTight = {\end{enumerate}},
    image = {

      % everything uses \linewidth: measuring against \postnarrowwidth again

      \postblock{%
        \begin{minipage}{\postnarrowwidth}
        \centering
        \ifpostgroupmode
          \setlength{\postimageheight}{0.4\textheight}%
        \else
          \setlength{\postimageheight}{0.75\textheight}%
        \fi
        \if\relax\detokenize{#1}\relax
          \setlength{\postcaptionheight}{0pt}%
        \else
          \sbox{\postcaptionbox}{%
            \begin{minipage}{\linewidth}%
              \raggedright
              \small
              #1%
            \end{minipage}%
          }%
          \setlength{\postcaptionheight}{\dimexpr\ht\postcaptionbox
            +\dp\postcaptionbox+\postblockgap\relax}%
        \fi
        \fitbox[
          maxwidth=\linewidth,
          maxheight=\postimageheight,
          belowboxspace=\postcaptionheight,
        ]{%
          \includegraphics{\postimagedir#3}%
        }%
        \par
        \postcaption{#1}%
        \end{minipage}%
      }%
    },
    link = {\posthref{#1}{#3}},
    emphasis = {{\itshape #1}},
    strongEmphasis = {{\bfseries \postbreakables{#1}}},
    % markdown delivers a literal backslash as the `backslash` renderer
    % (expanding to \textbackslash); append a near-forbidden penalty so a
    % Windows path can break after its separators only as a last resort.
    backslash = {\textbackslash{}\penalty9990},
    headingOne = {\postheadingone{#1}},
    headingTwo = {\postheadingtwo{#1}},
    headingThree = {\postheadingthree{#1}},
    headingFour = {\postheadingfour{#1}},
    headingFive = {\postheadingfive{#1}},
    headingSix = {\postheadingsix{#1}},
    blockQuoteBegin = {\postblockquotebegin},
    blockQuoteEnd = {\postblockquoteend},
    inputFencedCode = {\postinputfencedcode{#1}{#2}},
    codeSpan = {\postcodespan{#1}},
    note = {\postendnote{#1}},
  }
}
